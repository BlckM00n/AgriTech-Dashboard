Código C++ para Arduino UNO R4 WiFi

Para conectar tu placa física Arduino UNO R4 WiFi con este dashboard, sube el siguiente sketch preparado con las librerías WiFiS3, PubSubClient y LiquidCrystal_I2C.

Instrucciones rápidas:

1. Instala las librerías PubSubClient y LiquidCrystal_I2C en el Gestor de Librerías de tu Arduino IDE.

2. Rellena tus credenciales Wi-Fi (WIFI_SSID y WIFI_CLAVE) en el sketch.

3. Compila y sube el código a tu Arduino UNO R4 WiFi. Al iniciar, comenzará a transmitir los datos de tus sensores al tópico duoc/siy6122/fertirriego/datos.

/*
  FERTIRRIEGO IoT - Arduino UNO R4 WiFi (Modo Híbrido: Con/Sin WiFi)
  SIY6122 - Problemáticas Globales y Prototipado
  4 sensores analógicos -> decisión -> LCD 16x2 I2C (5 s por dato) + MQTT unificado
*/

#include <WiFiS3.h>
#include <PubSubClient.h>
#include <LiquidCrystal_I2C.h>

// ---------- Configuración ----------
const char* WIFI_SSID  = "TU_WIFI";
const char* WIFI_CLAVE = "TU_CLAVE";
const char* BROKER     = "broker.hivemq.com";
const char* TEMA       = "duoc/siy6122/fertirriego/datos"; // Tópico único con JSON completo
const char* ID_CLIENTE = "nodo-fertirriego-grupo01";   // debe ser único

const int   PINES[4]   = {A0, A1, A2, A3};
const char* NOMBRES[4] = {"temp", "humedad", "ph", "ce"};
const char* UNID[4]    = {"C", "%", "", "dS/m"};

// ---------- Tiempos (ms) ----------
const unsigned long T_LECTURA   = 1000;    // leer sensores y refrescar LCD/Serial
const unsigned long T_PAGINA    = 5000;    // cada dato visible 5 segundos
const unsigned long T_MQTT      = 3000;    // publicación MQTT
const unsigned long T_REINT_WIFI = 30000;  // reintento de WiFi
const unsigned long T_REINT_MQTT = 10000;  // reintento de broker

// ---------- Calibración de Lectura Analógica (0 - 1023) ----------
const bool PH_CE_REALES = false;  // false: potenciómetros simulan pH y CE

LiquidCrystal_I2C lcd(0x27, 16, 2);
WiFiClient red;
PubSubClient mqtt(red);

float valores[4];
const char* estados[4];

int pagina = 0;
unsigned long tLectura = 0, tPagina = 0, tMqtt = 0;
unsigned long tIntentoWiFi = 0, tIntentoMqtt = 0;
bool wifiIntentado = false;

// ---------- Captura de datos (Mapeo directo y sin ruido) ----------
float leer(int i) {
  int raw = analogRead(PINES[i]);
  switch (i) {
    case 0:  return (raw / 1023.0) * 50.0;                    // Temperatura: 0 a 50 °C
    case 1:  return (raw / 1023.0) * 100.0;                   // Humedad: 0 a 100 %
    case 2:  return (raw / 1023.0) * 14.0;                    // pH: 0 a 14
    default: return (raw / 1023.0) * 3.0;                     // CE: 0 a 3.0 mS/cm
  }
}

// ---------- Toma de decisiones ----------
const char* estado(int i, float v) {
  switch (i) {
    case 0:  return v >= 35 ? "ALERTA TEMP ALTA" : "Estado: NORMAL";
    case 1:  return v < 30 ? "! ESTRES HIDRICO" : v > 80 ? "! EXCESO DE AGUA" : "Humedad OK";
    case 2:  return v < 5 ? "! ALCALINIZAR" : v > 7.2 ? "! ACIDIFICAR"
                  : (v >= 5.5 && v <= 6.5) ? "RANGO OPTIMO" : "FUERA DE OPTIMO";
    default: return v > 3 ? "ALERTA: Salino"
                  : (v >= 1.5 && v <= 2.5) ? "Rango: Optimo" : "Fuera de Optimo";
  }
}

// ---------- LCD sin parpadeo ----------
String ajustar(String s, int largo) {
  while ((int)s.length() < largo) s += ' ';
  return s.substring(0, largo);
}

void dibujarLCD() {
  int i = pagina;
  String l0 = String(NOMBRES[i]) + ": " + String(valores[i], 1) + UNID[i];
  lcd.setCursor(0, 0);
  lcd.print(ajustar(l0, 15));
  lcd.print(mqtt.connected() ? '*' : '-');     // * = MQTT activo | - = sin red/MQTT
  lcd.setCursor(0, 1);
  lcd.print(ajustar(estados[i], 16));
}

// ---------- Red no bloqueante ----------
void gestionarRed() {
  if (WiFi.status() != WL_CONNECTED) {
    if (!wifiIntentado || millis() - tIntentoWiFi > T_REINT_WIFI) {
      tIntentoWiFi = millis();
      wifiIntentado = true;
      WiFi.begin(WIFI_SSID, WIFI_CLAVE);
    }
    return;
  }
  if (!mqtt.connected()) {
    if (millis() - tIntentoMqtt > T_REINT_MQTT) {
      tIntentoMqtt = millis();
      mqtt.connect(ID_CLIENTE);
    }
  } else {
    mqtt.loop();
  }
}

// ---------- Publicación MQTT unificada y sin errores de sintaxis ----------
void publicarMQTT() {
  if (!mqtt.connected()) return;

  // Construcción segura del paquete JSON para el dashboard
  String msg = "{";
  msg += "\"temp\":";
  msg += String(valores[0], 2);
  msg += ",\"humedad\":";
  msg += String(valores[1], 2);
  msg += ",\"pH\":";
  msg += String(valores[2], 2);
  msg += ",\"CE\":";
  msg += String(valores[3], 2);
  msg += ",\"nivelAgua\":80";
  msg += "}";

  mqtt.publish(TEMA, msg.c_str());
}

void setup() {
  Serial.begin(115200);
  lcd.init();
  lcd.backlight();
  lcd.print("Fertirriego IoT");
  lcd.setCursor(0, 1);
  lcd.print("Modo Hibrido");

  mqtt.setServer(BROKER, 1883);

  // Primera lectura inicial
  for (int i = 0; i < 4; i++) { valores[i] = leer(i); estados[i] = estado(i, valores[i]); }
  delay(1500);
  tLectura = tPagina = tMqtt = millis();
  dibujarLCD();
}

void loop() {
  gestionarRed();
  unsigned long ahora = millis();

  // 1) Lectura de sensores cada 1 s
  if (ahora - tLectura >= T_LECTURA) {
    tLectura = ahora;
    for (int i = 0; i < 4; i++) {
      valores[i] = leer(i);
      estados[i] = estado(i, valores[i]);
      Serial.print(NOMBRES[i]); Serial.print(": "); Serial.print(valores[i], 2);
      Serial.print(UNID[i]);    Serial.print(" | "); Serial.println(estados[i]);
    }
    Serial.println("---");
    dibujarLCD();
  }

  // 2) Cambio de página en pantalla LCD cada 5 s
  if (ahora - tPagina >= T_PAGINA) {
    tPagina = ahora;
    pagina = (pagina + 1) % 4;
    dibujarLCD();
  }

  // 3) MQTT cada 3 s
  if (ahora - tMqtt >= T_MQTT) {
    tMqtt = ahora;
    if (mqtt.connected()) publicarMQTT();
  }
}
